# Almost all of this app is behind a login, so nothing below is secret -- a crawler
# that ignores this file gets a 302 to the login form and learns nothing. What it costs
# without this file is the request, the redirect, and a line in the log, on every page
# of an account's history, for every crawler, forever. This is the cheapest way to say
# "there is nothing here for you".

User-agent: *

# The public surface, and all of it. Five pages and the root, which redirects to the
# first of them.
Allow: /$
Allow: /welcome
Allow: /about
Allow: /docs
Allow: /login
Allow: /create-account

# An account's own pages. /start is the empty-state guide and is behind a login too.
Disallow: /start
Disallow: /workouts
Disallow: /exercises
Disallow: /volume
Disallow: /programs
Disallow: /settings
Disallow: /equipment

# The OAuth endpoints, and the consent screen most of all: it is the one page in this
# app where a single tap hands an API client the account, so it is the last page that
# should ever be arrived at from a search result.
Disallow: /connections
Disallow: /authorize
Disallow: /token
Disallow: /oauth-

# Discovery metadata. Public by design and fetched by OAuth clients, which do not read
# this file; it is listed because a search index is not what it is public for.
Disallow: /.well-known/

# The Sitemap line this file has been holding open, and #359 is the day it predicted:
# documentation for the connector is indeed the first page it was waiting for.
#
# The original argument was that the public pages link to each other, so a sitemap would
# tell a crawler only what it already has. /docs is linked too -- from the front page and
# from /about, because a page nobody can find from the site is a bad page whatever a
# crawler thinks -- so that argument has not stopped being true, and this line is no longer
# load-bearing for discovery.
#
# It goes in anyway, for the reason that has changed rather than the one that has not:
# /docs is the URL a directory submission points at and the one page here that is aimed at
# a search term this domain can win, so it is the one page whose indexing is worth being
# explicit about rather than left to a crawl finding its way down from the root.
Sitemap: https://tectonicplates.app/sitemap.xml
